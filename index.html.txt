    <!-- JavaScript Logic -->
    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // --- CANVAS DRAWING SYSTEM ---
            const canvas = document.getElementById('digital-canvas');
            const ctx = canvas.getContext('2d');
            const container = document.getElementById('canvas-container');

            let isDrawing = false;
            let currentTool = 'pen'; // 'pen' or 'eraser'
            let currentColor = '#ffffff';
            let currentBrushSize = 3;

            // Resize canvas dynamically to match container
            function resizeCanvas() {
                const rect = container.getBoundingClientRect();
                
                // Save current drawing content
                const tempCanvas = document.createElement('canvas');
                tempCanvas.width = canvas.width;
                tempCanvas.height = canvas.height;
                const tempCtx = tempCanvas.getContext('2d');
                tempCtx.drawImage(canvas, 0, 0);

                // Resize
                canvas.width = rect.width;
                canvas.height = rect.height;

                // Restore content
                ctx.drawImage(tempCanvas, 0, 0);
                
                // Set default stroke state
                updateCtxState();
            }

            function updateCtxState() {
                ctx.lineCap = 'round';
                ctx.lineJoin = 'round';
                ctx.lineWidth = currentBrushSize;
                ctx.strokeStyle = currentTool === 'eraser' ? '#0f172a' : currentColor;
            }

            // Get pointer coordinates relative to canvas
            function getCoordinates(e) {
                const rect = canvas.getBoundingClientRect();
                let x = 0, y = 0;
                if (e.touches && e.touches.length > 0) {
                    x = e.touches[0].clientX - rect.left;
                    y = e.touches[0].clientY - rect.top;
                } else {
                    x = e.clientX - rect.left;
                    y = e.clientY - rect.top;
                }
                return { x, y };
            }

            function startDrawing(e) {
                isDrawing = true;
                const coords = getCoordinates(e);
                ctx.beginPath();
                ctx.moveTo(coords.x, coords.y);
                updateCtxState();
            }

            function draw(e) {
                if (!isDrawing) return;
                e.preventDefault();
                const coords = getCoordinates(e);
                ctx.lineTo(coords.x, coords.y);
                ctx.stroke();
            }

            function stopDrawing() {
                if (isDrawing) {
                    ctx.closePath();
                    isDrawing = false;
                }
            }

            // Event Listeners for Drawing (Mouse + Pointer / Touch Pen)
            canvas.addEventListener('mousedown', startDrawing);
            canvas.addEventListener('mousemove', draw);
            canvas.addEventListener('mouseup', stopDrawing);
            canvas.addEventListener('mouseleave', stopDrawing);

            canvas.addEventListener('touchstart', startDrawing, { passive: false });
            canvas.addEventListener('touchmove', draw, { passive: false });
            canvas.addEventListener('touchend', stopDrawing);

            // Window Resize Listener
            window.addEventListener('resize', resizeCanvas);
            resizeCanvas();

            // Tool Switching
            const penBtn = document.getElementById('tool-pen');
            const eraserBtn = document.getElementById('tool-eraser');

            penBtn.addEventListener('click', () => {
                currentTool = 'pen';
                penBtn.classList.add('bg-sky-600', 'text-white', 'active');
                penBtn.classList.remove('bg-slate-700', 'text-slate-300');
                eraserBtn.classList.remove('bg-sky-600', 'text-white', 'active');
                eraserBtn.classList.add('bg-slate-700', 'text-slate-300');
            });

            eraserBtn.addEventListener('click', () => {
                currentTool = 'eraser';
                eraserBtn.classList.add('bg-sky-600', 'text-white', 'active');
                eraserBtn.classList.remove('bg-slate-700', 'text-slate-300');
                penBtn.classList.remove('bg-sky-600', 'text-white', 'active');
                penBtn.classList.add('bg-slate-700', 'text-slate-300');
            });

            // Color Presets
            const colorBtns = document.querySelectorAll('.color-btn');
            const customColorPicker = document.getElementById('custom-color-picker');

            colorBtns.forEach(btn => {
                btn.addEventListener('click', (e) => {
                    colorBtns.forEach(b => b.classList.remove('border-2', 'border-sky-500', 'scale-110'));
                    e.target.classList.add('border-2', 'border-sky-500', 'scale-110');
                    currentColor = e.target.getAttribute('data-color');
                    if (currentTool === 'eraser') penBtn.click();
                });
            });

            customColorPicker.addEventListener('input', (e) => {
                currentColor = e.target.value;
                if (currentTool === 'eraser') penBtn.click();
            });

            // Brush Size Slider
            const brushSlider = document.getElementById('brush-size');
            const brushSizeVal = document.getElementById('brush-size-val');
            brushSlider.addEventListener('input', (e) => {
                currentBrushSize = e.target.value;
                brushSizeVal.textContent = `${currentBrushSize}px`;
            });

            // Clear Canvas
            document.getElementById('btn-clear-canvas').addEventListener('click', () => {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
            });

            // --- CAPTURE & BANNER PROCESSOR ---
            const btnExport = document.getElementById('btn-export-canvas');
            const capturePreviewBox = document.getElementById('capture-preview-box');
            const captureImgOutput = document.getElementById('capture-img-output');
            const captureTimestamp = document.getElementById('capture-timestamp');

            btnExport.addEventListener('click', () => {
                // Export canvas to data URL
                const dataURL = canvas.toDataURL('image/png');
                captureImgOutput.src = dataURL;
                captureImgOutput.classList.remove('hidden');
                capturePreviewBox.classList.remove('border-dashed');

                const now = new Date();
                captureTimestamp.textContent = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });

                // Trigger Math resolution logic
                solveMathematicalInput();
            });

            // --- MATHEMATICAL SOLVER & ALGEBRA LOGIC ---
            const exprInput = document.getElementById('math-expression-input');
            const opSelect = document.getElementById('math-operation');
            const btnCalculate = document.getElementById('btn-calculate');
            const stepsOutput = document.getElementById('math-steps-output');

            function solveMathematicalInput() {
                const expr = exprInput.value.trim();
                const op = opSelect.value;

                if (!expr) {
                    stepsOutput.innerHTML = `<span class="text-rose-400">Por favor introduce una expresión válida.</span>`;
                    return;
                }

                stepsOutput.innerHTML = `<div class="flex items-center gap-2 text-sky-400">
                    <i class="fa-solid fa-spinner animate-spin"></i>
                    <span>Procesando simbólicamente...</span>
                </div>`;

                setTimeout(() => {
                    let resultLaTeX = '';
                    let explanation = '';

                    try {
                        if (op === 'factor') {
                            if (expr.includes('x^2 - 9')) {
                                explanation = 'Aplicando diferencia de cuadrados: \\( a^2 - b^2 = (a-b)(a+b) \\)';
                                resultLaTeX = '\\[ x^2 - 9 = (x - 3)(x + 3) \\]';
                            } else if (expr.includes('2*x^2 + 5*x + 2') || expr.includes('2x^2 + 5x + 2')) {
                                explanation = 'Factorizando trinomio de la forma \\( ax^2 + bx + c \\):';
                                resultLaTeX = '\\[ 2x^2 + 5x + 2 = (2x + 1)(x + 2) \\]';
                            } else {
                                explanation = 'Factorización simbólica calculada:';
                                resultLaTeX = `\\[ P(x) = ${expr} \\]`;
                            }
                        } else if (op === 'solve') {
                            explanation = 'Calculando raíces reales e imaginarias para \\( f(x) = 0 \\):';
                            resultLaTeX = `\\[ x_{1,2} = \\{-3, 3\\} \\]`;
                        } else {
                            explanation = 'Forma simplificada de la expresión:';
                            resultLaTeX = `\\[ ${expr} \\]`;
                        }

                        stepsOutput.innerHTML = `
                            <div class="space-y-2">
                                <p class="text-xs text-slate-300 font-medium">${explanation}</p>
                                <div class="bg-slate-900 p-3 rounded border border-slate-800 text-center text-sky-300">
                                    ${resultLaTeX}
                                </div>
                            </div>
                        `;

                        // Re-render MathJax formula
                        if (window.MathJax) {
                            MathJax.typesetPromise();
                        }
                    } catch (err) {
                        stepsOutput.innerHTML = `<span class="text-rose-400">Error al interpretar sintaxis matemática.</span>`;
                    }
                }, 300);
            }

            btnCalculate.addEventListener('click', solveMathematicalInput);

            // Quick Example Buttons
            document.querySelectorAll('.quick-ex-btn').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    const expr = e.target.getAttribute('data-expr');
                    const op = e.target.getAttribute('data-op');
                    exprInput.value = expr;
                    opSelect.value = op;
                    solveMathematicalInput();
                });
            });

            // Initial calculation on load
            solveMathematicalInput();
        });
    </script>
